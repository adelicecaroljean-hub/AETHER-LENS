local function updateCursorState()
	if freecamActive and (isRightMouseDown or touchPanning) then
		UserInputService.MouseBehavior = Enum.MouseBehavior.LockCenter
		UserInputService.MouseIconEnabled = false
	else
		UserInputService.MouseBehavior = Enum.MouseBehavior.Default
		UserInputService.MouseIconEnabled = true
	end
end

local function setCharacterControls(enabled)
	local character = LocalPlayer.Character
	if character then
		local humanoid = character:FindFirstChildOfClass("Humanoid")
		local rootPart = character:FindFirstChild("HumanoidRootPart")

		if humanoid then
			humanoid:Move(Vector3.zero, false)
		end

		if rootPart then
			rootPart.AssemblyLinearVelocity = Vector3.zero
			rootPart.AssemblyAngularVelocity = Vector3.zero
			rootPart.Anchored = not enabled
		end
	end

	local playerScripts = LocalPlayer:FindFirstChild("PlayerScripts")
	if playerScripts then
		local playerModuleScript = playerScripts:FindFirstChild("PlayerModule")
		if playerModuleScript then
			local ok, playerModule = pcall(require, playerModuleScript)
			if ok and playerModule then
				local controls = playerModule:GetControls()
				if controls then
					if enabled then controls:Enable() else controls:Disable() end
				end
			end
		end
	end
end

local function attachButtonFx(btn, baseColor, hoverColor, strokeObj)
	btn.MouseEnter:Connect(function()
		TweenService:Create(btn, TweenInfo.new(0.18), {BackgroundColor3 = hoverColor}):Play()
		if strokeObj then
			TweenService:Create(strokeObj, TweenInfo.new(0.18), {Color = Color3.fromRGB(0, 195, 255)}):Play()
		end
	end)
	btn.MouseLeave:Connect(function()
		TweenService:Create(btn, TweenInfo.new(0.18), {BackgroundColor3 = baseColor}):Play()
		if strokeObj then
			TweenService:Create(strokeObj, TweenInfo.new(0.18), {Color = Color3.fromRGB(45, 52, 68)}):Play()
		end
	end)
	btn.MouseButton1Down:Connect(function()
		TweenService:Create(btn, TweenInfo.new(0.08), {
			Size = UDim2.new(btn.Size.X.Scale, btn.Size.X.Offset - 4, btn.Size.Y.Scale, btn.Size.Y.Offset - 2)
		}):Play()
	end)
	btn.MouseButton1Up:Connect(function()
		TweenService:Create(btn, TweenInfo.new(0.12, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
			Size = UDim2.new(btn.Size.X.Scale, btn.Size.X.Offset, btn.Size.Y.Scale, btn.Size.Y.Offset)
		}):Play()
	end)
end

attachButtonFx(langBtn, Color3.fromRGB(28, 32, 42), Color3.fromRGB(36, 42, 56), langStroke)

-- Header Dragging
local draggingWindow = false
local dragStartPos, frameStartPos

titleBar.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		draggingWindow = true
		dragStartPos = input.Position
		frameStartPos = mainFrame.Position

		local connection
		connection = input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				draggingWindow = false
				connection:Disconnect()
			end
		end)
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if draggingWindow and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStartPos
		mainFrame.Position = UDim2.new(
			frameStartPos.X.Scale, 
			frameStartPos.X.Offset + delta.X, 
			frameStartPos.Y.Scale, 
			frameStartPos.Y.Offset + delta.Y
		)
	end
end)

local function createSettingRow(order, localeKey, minVal, maxVal, defaultVal, callback)
	local row = Instance.new("Frame")
	row.Size = UDim2.new(1, 0, 0, 50)
	row.BackgroundColor3 = Color3.fromRGB(24, 27, 35)
	row.BorderSizePixel = 0
	row.LayoutOrder = order
	row.Parent = container

	local rowCorner = Instance.new("UICorner")
	rowCorner.CornerRadius = UDim.new(0, 8)
	rowCorner.Parent = row

	local rowStroke = Instance.new("UIStroke")
	rowStroke.Color = Color3.fromRGB(35, 40, 52)
	rowStroke.Thickness = 1
	rowStroke.Parent = row

	local title = Instance.new("TextLabel")
	title.Size = UDim2.new(0.65, 0, 0, 20)
	title.Position = UDim2.new(0, 10, 0, 4)
	title.BackgroundTransparency = 1
	title.Text = LOCALES[currentLang][localeKey]
	title.TextColor3 = Color3.fromRGB(215, 220, 230)
	title.TextSize = 10
	title.Font = Enum.Font.GothamMedium
	title.TextXAlignment = Enum.TextXAlignment.Left
	title.Parent = row
	textElements["row_" .. localeKey] = title

	local valLabel = Instance.new("TextLabel")
	valLabel.Size = UDim2.new(0.30, 0, 0, 20)
	valLabel.Position = UDim2.new(0.68, 0, 0, 4)
	valLabel.BackgroundTransparency = 1
	valLabel.Text = tostring(defaultVal)
	valLabel.TextColor3 = Color3.fromRGB(0, 195, 255)
	valLabel.TextSize = 10
	valLabel.Font = Enum.Font.GothamBold
	valLabel.TextXAlignment = Enum.TextXAlignment.Right
	valLabel.Parent = row

	local track = Instance.new("Frame")
	track.Size = UDim2.new(1, -20, 0, 6)
	track.Position = UDim2.new(0, 10, 0, 31)
	track.BackgroundColor3 = Color3.fromRGB(38, 44, 58)
	track.BorderSizePixel = 0
	track.Parent = row

	local trackCorner = Instance.new("UICorner")
	trackCorner.CornerRadius = UDim.new(1, 0)
	trackCorner.Parent = track

	local fill = Instance.new("Frame")
	local initAlpha = math.clamp((defaultVal - minVal) / (maxVal - minVal), 0, 1)
	fill.Size = UDim2.new(initAlpha, 0, 1, 0)
	fill.BackgroundColor3 = Color3.fromRGB(0, 195, 255)
	fill.BorderSizePixel = 0
	fill.Parent = track

	local fillGrad = Instance.new("UIGradient")
	fillGrad.Color = ColorSequence.new({
		ColorSequenceKeypoint.new(0, Color3.fromRGB(0, 150, 255)),
		ColorSequenceKeypoint.new(1, Color3.fromRGB(0, 220, 255))
	})
	fillGrad.Parent = fill

	row.MouseEnter:Connect(function()
		TweenService:Create(row, TweenInfo.new(0.18), {BackgroundColor3 = Color3.fromRGB(28, 32, 42)}):Play()
		TweenService:Create(rowStroke, TweenInfo.new(0.18), {Color = Color3.fromRGB(0, 195, 255)}):Play()
	end)
	row.MouseLeave:Connect(function()
		TweenService:Create(row, TweenInfo.new(0.18), {BackgroundColor3 = Color3.fromRGB(24, 27, 35)}):Play()
		TweenService:Create(rowStroke, TweenInfo.new(0.18), {Color = Color3.fromRGB(35, 40, 52)}):Play()
	end)

	local dragging = false

	local function update(input)
		local x = math.clamp((input.Position.X - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
		local value = minVal + (maxVal - minVal) * x
		
		if (maxVal - minVal) <= 2 then
			value = math.floor(value * 100 + 0.5) / 100
		else
			value = math.floor(value + 0.5)
		end

		TweenService:Create(fill, TweenInfo.new(0.08, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
			Size = UDim2.new(x, 0, 1, 0)
		}):Play()

		valLabel.Text = tostring(value)
		callback(value)
	end

	track.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			update(input)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			update(input)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
end

	barFill.BackgroundColor3 = Color3.fromRGB(0, 195, 255)
	barFill.BorderSizePixel = 0
	barFill.ZIndex = 102
	barFill.Parent = barBg

	local fillGrad = Instance.new("UIGradient")
	fillGrad.Color = ColorSequence.new({
		ColorSequenceKeypoint.new(0, Color3.fromRGB(0, 140, 255)),
		ColorSequenceKeypoint.new(1, Color3.fromRGB(0, 230, 255))
	})
	fillGrad.Parent = barFill

	local steps = LOCALES[currentLang].loadSteps
	local targets = {0.20, 0.48, 0.75, 0.92, 1.00}
	local durations = {0.8, 0.9, 0.9, 0.7, 0.5}

	local currentProgress = 0

	for i = 1, #steps do
		statusLabel.Text = steps[i]
		local startVal = currentProgress
		local endVal = targets[i]
		local dur = durations[i]

		TweenService:Create(barFill, TweenInfo.new(dur, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out), {
			Size = UDim2.new(endVal, 0, 1, 0)
		}):Play()

		local startTime = tick()
		while (tick() - startTime) < dur do
			local alpha = (tick() - startTime) / dur
			local val = math.floor((startVal + (endVal - startVal) * alpha) * 100)
			percentLabel.Text = tostring(math.clamp(val, 0, 100)) .. "%"
			task.wait()
		end

		currentProgress = endVal
		percentLabel.Text = tostring(math.floor(endVal * 100)) .. "%"
	end

	task.wait(0.3)

	local flashFrame = Instance.new("Frame")
	flashFrame.Size = UDim2.new(1, 0, 1, 0)
	flashFrame.BackgroundColor3 = Color3.fromRGB(0, 195, 255)
	flashFrame.BackgroundTransparency = 1
	flashFrame.ZIndex = 110
	flashFrame.Parent = screenGui

	TweenService:Create(flashFrame, TweenInfo.new(0.15), {BackgroundTransparency = 0.85}):Play()
	task.wait(0.15)
	TweenService:Create(flashFrame, TweenInfo.new(0.35), {BackgroundTransparency = 1}):Play()

	if spinConnection then spinConnection:Disconnect() end

	local fadeInfo = TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
	TweenService:Create(loadBlur, fadeInfo, {Size = 0}):Play()
	TweenService:Create(loadFrame, fadeInfo, {BackgroundTransparency = 1}):Play()
	TweenService:Create(loadLogo, fadeInfo, {ImageTransparency = 1}):Play()
	TweenService:Create(logoGlow, fadeInfo, {ImageTransparency = 1}):Play()
	TweenService:Create(ringStroke, fadeInfo, {Transparency = 1}):Play()
	TweenService:Create(titleLabel, fadeInfo, {TextTransparency = 1}):Play()
	TweenService:Create(verBadge, fadeInfo, {TextTransparency = 1}):Play()
	TweenService:Create(statusLabel, fadeInfo, {TextTransparency = 1}):Play()
	TweenService:Create(percentLabel, fadeInfo, {TextTransparency = 1}):Play()
	TweenService:Create(barBg, fadeInfo, {BackgroundTransparency = 1}):Play()
	TweenService:Create(barFill, fadeInfo, {BackgroundTransparency = 1}):Play()

	task.wait(0.5)
	loadFrame:Destroy()
	loadBlur:Destroy()
	flashFrame:Destroy()

	revealMainMenu()
end

local function setFreecamState(active)
	freecamActive = active

	for k in pairs(keysPressed) do
		keysPressed[k] = false
	end

	if freecamActive then
		toggleBtn.Text = LOCALES[currentLang].btnActive
		TweenService:Create(toggleBtn, TweenInfo.new(0.2), {
			BackgroundColor3 = Color3.fromRGB(0, 160, 230),
			TextColor3 = Color3.fromRGB(255, 255, 255)
		}):Play()

		if bgmSound then
			TweenService:Create(bgmSound, TweenInfo.new(0.5), {Volume = settings.bgmVolume * 0.25}):Play()
		end

		camera.CameraType = Enum.CameraType.Scriptable
		cameraPos = camera.CFrame.Position
		targetCFrame = camera.CFrame
		currentCFrame = camera.CFrame

		local rx, ry, _ = camera.CFrame:ToOrientation()
		pitch = math.deg(rx)
		yaw = math.deg(ry)

		setCharacterControls(false)
		updateCursorState()
	else
		toggleBtn.Text = LOCALES[currentLang].btnEnable
		TweenService:Create(toggleBtn, TweenInfo.new(0.2), {
			BackgroundColor3 = Color3.fromRGB(28, 32, 42),
			TextColor3 = Color3.fromRGB(210, 215, 225)
		}):Play()

		if bgmSound then
			TweenService:Create(bgmSound, TweenInfo.new(0.5), {Volume = settings.bgmVolume}):Play()
		end

		camera.CameraType = Enum.CameraType.Custom
		if LocalPlayer.Character then
			local humanoid = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
			if humanoid then
				camera.CameraSubject = humanoid
			end
		end

		isRightMouseDown = false
		touchPanning = false
		setCharacterControls(true)
		updateCursorState()
	end
end

local function toggleFreecam()
	setFreecamState(not freecamActive)
end

local function toggleGuiVisibility()
	uiVisible = not uiVisible
	mainFrame.Visible = uiVisible
end

toggleBtn.MouseButton1Click:Connect(toggleFreecam)

-- Keyboard & Touch Input Handlers
UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end

	if input.KeyCode == settings.toggleKey then
		toggleFreecam()
	elseif input.KeyCode == settings.uiToggleKey then
		toggleGuiVisibility()
	elseif freecamActive then
		if input.KeyCode == Enum.KeyCode.W then keysPressed.W = true end
		if input.KeyCode == Enum.KeyCode.S then keysPressed.S = true end
		if input.KeyCode == Enum.KeyCode.A then keysPressed.A = true end
		if input.KeyCode == Enum.KeyCode.D then keysPressed.D = true end
		if input.KeyCode == Enum.KeyCode.E then keysPressed.E = true end
		if input.KeyCode == Enum.KeyCode.Q then keysPressed.Q = true end
		if input.KeyCode == Enum.KeyCode.LeftShift then keysPressed.Shift = true end
	end

	-- Right Mouse Look
	if input.UserInputType == Enum.UserInputType.MouseButton2 and freecamActive then
		isRightMouseDown = true
		updateCursorState()
	end

	-- Mobile Touch Pan
	if input.UserInputType == Enum.UserInputType.Touch and freecamActive then
		touchPanning = true
		touchStartPos = Vector2.new(input.Position.X, input.Position.Y)
		updateCursorState()
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.KeyCode == Enum.KeyCode.W then keysPressed.W = false end
	if input.KeyCode == Enum.KeyCode.S then keysPressed.S = false end
	if input.KeyCode == Enum.KeyCode.A then keysPressed.A = false end
	if input.KeyCode == Enum.KeyCode.D then keysPressed.D = false end
	if input.KeyCode == Enum.KeyCode.E then keysPressed.E = false end
	if input.KeyCode == Enum.KeyCode.Q then keysPressed.Q = false end
	if input.KeyCode == Enum.KeyCode.LeftShift then keysPressed.Shift = false end

	if input.UserInputType == Enum.UserInputType.MouseButton2 then
		isRightMouseDown = false
		updateCursorState()
	end

	if input.UserInputType == Enum.UserInputType.Touch then
		touchPanning = false
		updateCursorState()
	end
end)

-- Mouse & Touch Camera Look Delta Computation
UserInputService.InputChanged:Connect(function(input)
	if not freecamActive then return end

	if isRightMouseDown and input.UserInputType == Enum.UserInputType.MouseMovement then
		yaw = yaw - input.Delta.X * settings.sensitivity
		pitch = math.clamp(pitch - input.Delta.Y * settings.sensitivity, -89, 89)
	elseif touchPanning and input.UserInputType == Enum.UserInputType.Touch then
		local currentPos = Vector2.new(input.Position.X, input.Position.Y)
		local delta = currentPos - touchStartPos
		touchStartPos = currentPos

		yaw = yaw - delta.X * (settings.sensitivity * 0.8)
		pitch = math.clamp(pitch - delta.Y * (settings.sensitivity * 0.8), -89, 89)
	end
end)

-- Render Update Loop
RunService.RenderStepped:Connect(function(dt)
	if not freecamActive then return end

	local rotCFrame = CFrame.Angles(0, math.rad(yaw), 0) * CFrame.Angles(math.rad(pitch), 0, 0)

	local moveDir = Vector3.zero
	if keysPressed.W then moveDir += Vector3.new(0, 0, -1) end
	if keysPressed.S then moveDir += Vector3.new(0, 0, 1) end
	if keysPressed.A then moveDir += Vector3.new(-1, 0, 0) end
	if keysPressed.D then moveDir += Vector3.new(1, 0, 0) end
	if keysPressed.E then moveDir += Vector3.new(0, 1, 0) end
	if keysPressed.Q then moveDir += Vector3.new(0, -1, 0) end

	if moveDir.Magnitude > 0 then
		moveDir = moveDir.Unit
	end

	local speed = settings.baseSpeed * (keysPressed.Shift and settings.sprintMult or 1)
	local worldDelta = rotCFrame:VectorToWorldSpace(moveDir) * (speed * dt)

	cameraPos = cameraPos + worldDelta
	targetCFrame = CFrame.new(cameraPos) * rotCFrame

	local alpha = math.clamp(dt * (1 / settings.smoothness) * 10, 0, 1)
	currentCFrame = currentCFrame:Lerp(targetCFrame, alpha)
	camera.CFrame = currentCFrame

	camera.FieldOfView = camera.FieldOfView + (settings.fov - camera.FieldOfView) * alpha
end)
